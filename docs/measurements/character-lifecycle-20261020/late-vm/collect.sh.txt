#!/bin/bash
set -eu
ROOT=/Users/nyatinte/orca/workspaces/nyatinte-animation/loach
cd "$ROOT"
OUT="$ROOT/dist/memory/late-vm-98-20261020"
mkdir -p "$OUT"
REVISION=b11e696e2dc673ae46aed73f675226527a608f45
SHORT=$(git rev-parse --short "$REVISION")
touch "$OUT/start.marker"
{
  date -u
  printf 'source_revision=%s\nProduction Waddly PID 35688 is left untouched; ambient activity is uncontrolled.\n' "$REVISION"
  memory_pressure -Q
  vm_stat
} > "$OUT/context.txt"
WADDLY_PROFILE_CHARACTER_LIFECYCLE=1 WADDLY_PROFILE_CHARACTER_CYCLES=10 \
WADDLY_PROFILE_STACK_LOGGING=0 WADDLY_PROFILE_IMAGE_FIXTURE='' \
./macos/profile-image-app.sh "$REVISION" > "$OUT/probe.output.txt" 2>&1 &
WRAPPER=$!
PROBE=''
DONE=0
cleanup() {
  if [ "$DONE" -eq 0 ]; then
    if [ -n "$PROBE" ]; then kill -TERM "$PROBE" 2>/dev/null || true; fi
    kill -TERM "$WRAPPER" 2>/dev/null || true
  fi
}
trap cleanup EXIT
DEADLINE=$((SECONDS + 900))
CSV=''
while [ -z "$CSV" ]; do
  test "$SECONDS" -lt "$DEADLINE"
  kill -0 "$WRAPPER"
  CSV=$(find "$ROOT/dist/memory" -maxdepth 1 -newer "$OUT/start.marker" -name "image-app-$SHORT-*.csv" | head -1)
  sleep 1
done
PROBE=$(pgrep -f '^/[^ ]*/WaddlyMemoryProbe.app/Contents/MacOS/WaddlyMemoryProbe ')
case "$PROBE" in ''|*$'\n'*) printf 'Expected one isolated probe PID\n' >&2; exit 1;; esac
printf 'target_pid=%s\nwrapper_pid=%s\n' "$PROBE" "$WRAPPER" > "$OUT/inspection-status.txt"
for cycle in 0 2 7 9; do
  label="cycle-$cycle-closed-20s"
  while ! grep -q "^$label," "$CSV"; do
    test "$SECONDS" -lt "$DEADLINE"
    kill -0 "$WRAPPER"
    sleep 1
  done
  {
    printf 'requested_state=%s\ncommand=vmmap -summary %s\n' "$label" "$PROBE"
    date -u
    printf 'csv_before='; tail -1 "$CSV"
  } > "$OUT/cycle-$cycle.inspection.txt"
  if vmmap -summary "$PROBE" > "$OUT/cycle-$cycle.vmmap.txt" 2>&1; then
    status=0
  else
    status=$?
  fi
  {
    printf 'vmmap_exit_code=%s\n' "$status"
    date -u
    printf 'csv_after='; tail -1 "$CSV"
  } >> "$OUT/cycle-$cycle.inspection.txt"
  printf 'cycle=%s vmmap_exit_code=%s\n' "$cycle" "$status" >> "$OUT/inspection-status.txt"
  printf 'Inspected %s, vmmap exit %s\n' "$label" "$status"
done
if wait "$WRAPPER"; then status=0; else status=$?; fi
DONE=1
printf 'probe_wrapper_exit_code=%s\n' "$status" >> "$OUT/inspection-status.txt"
cp "$CSV" "$OUT/probe.csv"
cp "${CSV%.csv}.environment.txt" "$OUT/probe.environment.txt"
{
  date -u
  memory_pressure -Q
  vm_stat
} >> "$OUT/context.txt"
test "$status" -eq 0
tail -1 "$CSV" | grep '^lifecycle-complete,' >> "$OUT/inspection-status.txt"
printf 'DIAGNOSTIC_COMPLETED\n'
