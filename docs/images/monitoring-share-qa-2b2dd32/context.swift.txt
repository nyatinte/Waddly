import AppKit
import ApplicationServices
let bundle = "dev.nyatinte.waddly.qa111b.development"
guard let app = NSRunningApplication.runningApplications(withBundleIdentifier: bundle).first else { exit(1) }
let point = CGPoint(x: 1572, y: 969)
let windows = CGWindowListCopyWindowInfo([.optionOnScreenOnly, .excludeDesktopElements], kCGNullWindowID) as? [[String: Any]] ?? []
let top = windows.first { info in
    guard let bounds = info[kCGWindowBounds as String] as? [String: Any],
          let rect = CGRect(dictionaryRepresentation: bounds as CFDictionary),
          (info[kCGWindowAlpha as String] as? Double ?? 0) > 0 else { return false }
    return rect.contains(point)
}
guard top?[kCGWindowOwnerPID as String] as? Int == Int(app.processIdentifier) else {
    print("ABORT unexpected topmost owner"); exit(1)
}
print("verified_topmost_pid", app.processIdentifier, "window", top?[kCGWindowNumber as String] as Any)
for type in [CGEventType.rightMouseDown, .rightMouseUp] {
    CGEvent(mouseEventSource: nil, mouseType: type, mouseCursorPosition: point, mouseButton: .right)?.post(tap: .cghidEventTap)
    Thread.sleep(forTimeInterval: 0.1)
}
