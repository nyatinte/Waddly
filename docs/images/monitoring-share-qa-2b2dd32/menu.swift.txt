import AppKit
import ApplicationServices

func value(_ element: AXUIElement, _ name: String) -> CFTypeRef? {
    var result: CFTypeRef?
    AXUIElementCopyAttributeValue(element, name as CFString, &result)
    return result
}
let args = CommandLine.arguments
let bundle = "dev.nyatinte.waddly.qa111b.development"
guard let app = NSRunningApplication.runningApplications(withBundleIdentifier: bundle).first else { exit(1) }
let root = AXUIElementCreateApplication(app.processIdentifier)
guard let barValue = value(root, "AXExtrasMenuBar") else { print("No extras menu bar"); exit(1) }
let bar = barValue as! AXUIElement
let items = value(bar, kAXChildrenAttribute) as? [AXUIElement] ?? []
if args[1] == "click-status" {
    guard let item = items.first,
          let p = value(item, kAXPositionAttribute), let s = value(item, kAXSizeAttribute) else { exit(1) }
    var origin = CGPoint.zero
    var size = CGSize.zero
    AXValueGetValue(p as! AXValue, .cgPoint, &origin)
    AXValueGetValue(s as! AXValue, .cgSize, &size)
    let point = CGPoint(x: origin.x + size.width / 2, y: origin.y + size.height / 2)
    print("STATUS_RECT", origin, size)
    for type in [CGEventType.leftMouseDown, .leftMouseUp] {
        CGEvent(mouseEventSource: nil, mouseType: type, mouseCursorPosition: point, mouseButton: .left)?.post(tap: .cghidEventTap)
        Thread.sleep(forTimeInterval: 0.1)
    }
    Thread.sleep(forTimeInterval: 0.4)
}
if args[1] == "open" {
    guard let item = items.first else { exit(1) }
    print("Open menu", AXUIElementPerformAction(item, kAXPressAction as CFString).rawValue)
    Thread.sleep(forTimeInterval: 0.2)
}
var pressed = false
func walk(_ element: AXUIElement, _ depth: Int) {
    guard depth < 15 else { return }
    let title = value(element, kAXTitleAttribute) as? String ?? ""
    let role = value(element, kAXRoleAttribute) as? String ?? ""
    print(role, title)
    if ["press", "click-item"].contains(args[1]), !pressed, title == args[2], role == kAXMenuItemRole {
        pressed = true
        if args[1] == "press" {
            print("AXPress", title, AXUIElementPerformAction(element, kAXPressAction as CFString).rawValue)
        } else if let p = value(element, kAXPositionAttribute), let s = value(element, kAXSizeAttribute) {
            var origin = CGPoint.zero
            var size = CGSize.zero
            AXValueGetValue(p as! AXValue, .cgPoint, &origin)
            AXValueGetValue(s as! AXValue, .cgSize, &size)
            print("ITEM_RECT", title, origin, size)
            guard origin.x > 0, origin.y > 0, size.width > 0, size.height > 0 else { exit(1) }
            let point = CGPoint(x: origin.x + size.width / 2, y: origin.y + size.height / 2)
            for type in [CGEventType.leftMouseDown, .leftMouseUp] {
                CGEvent(mouseEventSource: nil, mouseType: type, mouseCursorPosition: point, mouseButton: .left)?.post(tap: .cghidEventTap)
                Thread.sleep(forTimeInterval: 0.1)
            }
        }
    }
    for child in value(element, kAXChildrenAttribute) as? [AXUIElement] ?? [] { walk(child, depth + 1) }
}
walk(bar, 0)
