import AppKit
import os
import WaddlyCore

private enum InjectedFailure: Error { case generation }

@main
@MainActor
struct ShareQA {
    static var controller: PetShareWindowController?
    static var suite: String?
    static func main() throws {
        let args = CommandLine.arguments
        guard args.count == 4 else { exit(64) }
        let root = URL(fileURLWithPath: args[1])
        let english = args[2] == "en"
        let app = NSApplication.shared
        app.setActivationPolicy(.regular)
        app.appearance = NSAppearance(named: args[3] == "light" ? .aqua : .darkAqua)
        let suiteName = "Waddly.QA111B.Share.\(UUID().uuidString)"
        guard let defaults = UserDefaults(suiteName: suiteName),
              let resources = Bundle(url: root.appendingPathComponent("macos")),
              let images = PetSpriteSheetImporter.frames(from: try Data(contentsOf: root.appendingPathComponent("assets/examples/nyatinte-bot-3x3.png"))) else { exit(65) }
        suite = suiteName
        let settings = AppSettings(defaults: defaults)
        settings.appLanguage = english ? .english : .japanese
        let localization = LocalizationController(settings: settings, resources: resources)
        let attempts = OSAllocatedUnfairLock(initialState: 0)
        var effects = PetShareEffects()
        effects.writeMovie = { images, url, cancellation in
            let attempt = attempts.withLock { $0 += 1; return $0 }
            if attempt == 1 {
                let release = root.appendingPathComponent("dist/qa111b/release-\(english ? "en" : "ja")-failure")
                let deadline = DispatchTime.now() + .seconds(60)
                while !FileManager.default.fileExists(atPath: release.path), DispatchTime.now() < deadline, !cancellation.isCancelled {
                    Thread.sleep(forTimeInterval: 0.1)
                }
                throw InjectedFailure.generation
            }
            try PetShareExporter.writeMovie(from: images, to: url, cancellation: cancellation)
        }
        // Deliberate effect failures: do not write the real clipboard or open an external composer.
        effects.copyImage = { _ in false }
        effects.publishMovie = { _, _ in nil }
        effects.openURL = { _ in false }
        let share = PetShareWindowController(images: images, localization: localization, effects: effects)
        share.onClose = {
            defaults.removePersistentDomain(forName: suiteName)
            app.terminate(nil)
        }
        controller = share
        Task { @MainActor in
            share.showWindow(nil)
            share.prepareAssets()
        }
        app.run()
    }
}
