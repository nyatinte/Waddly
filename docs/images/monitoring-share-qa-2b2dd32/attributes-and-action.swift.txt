import AppKit
import ApplicationServices

func value(_ element: AXUIElement, _ key: String) -> CFTypeRef? {
    var result: CFTypeRef?
    AXUIElementCopyAttributeValue(element, key as CFString, &result)
    return result
}
let args = CommandLine.arguments
let mode = args[1]
if mode == "context" {
    let point = CGPoint(x: Double(args[2])!, y: Double(args[3])!)
    for type in [CGEventType.rightMouseDown, .rightMouseUp] {
        CGEvent(mouseEventSource: nil, mouseType: type, mouseCursorPosition: point, mouseButton: .right)?.post(tap: .cghidEventTap)
        Thread.sleep(forTimeInterval: 0.1)
    }
} else if mode == "drag" || mode == "hover" {
    let start = CGPoint(x: Double(args[2])!, y: Double(args[3])!)
    let end = CGPoint(x: Double(args[4])!, y: Double(args[5])!)
    let source = CGEventSource(stateID: .hidSystemState)
    func topWindow(at point: CGPoint) -> [String: Any]? {
        let windows = CGWindowListCopyWindowInfo([.optionOnScreenOnly, .excludeDesktopElements], kCGNullWindowID) as? [[String: Any]] ?? []
        return windows.first { info in
            guard let dictionary = info[kCGWindowBounds as String] as? [String: Any],
                  let rect = CGRect(dictionaryRepresentation: dictionary as CFDictionary),
                  (info[kCGWindowAlpha as String] as? Double ?? 0) > 0 else { return false }
            return rect.contains(point)
        }
    }
    if mode == "drag", args.count > 8 {
        let owner = topWindow(at: start)
        guard owner?[kCGWindowOwnerPID as String] as? Int == Int(args[8]) else { print("ABORT: unexpected source owner"); exit(1) }
        let target = topWindow(at: end)
        guard target?[kCGWindowOwnerPID as String] as? Int == Int(args[7]) else { print("ABORT: unexpected target owner"); exit(1) }
    }
    func post(_ type: CGEventType, _ point: CGPoint) {
        CGEvent(mouseEventSource: source, mouseType: type, mouseCursorPosition: point, mouseButton: .left)?.post(tap: .cghidEventTap)
    }
    post(.mouseMoved, start)
    Thread.sleep(forTimeInterval: 0.2)
    if mode == "drag" { post(.leftMouseDown, start) }
    for step in 1...50 {
        let fraction = Double(step) / 50
        post(mode == "drag" ? .leftMouseDragged : .mouseMoved,
             CGPoint(x: start.x + (end.x-start.x)*fraction, y: start.y + (end.y-start.y)*fraction))
        Thread.sleep(forTimeInterval: 0.025)
    }
    Thread.sleep(forTimeInterval: Double(args[6]) ?? 1)
    if mode == "drag" {
        if args.count > 7, topWindow(at: end)?[kCGWindowOwnerPID as String] as? Int != Int(args[7]) {
            post(.leftMouseDragged, start)
            post(.leftMouseUp, start)
            print("ABORT: target owner changed; released at source")
            exit(1)
        }
        post(.leftMouseUp, end)
    }
} else {
    let app = NSRunningApplication.runningApplications(withBundleIdentifier: args[2]).first!
    let element = AXUIElementCreateApplication(app.processIdentifier)
    let windows = value(element, kAXWindowsAttribute) as? [AXUIElement] ?? []
    let window = windows.first { (value($0, kAXTitleAttribute) as? String ?? "").contains(args[3]) }!
    if mode == "state" {
        print("AXTitle", value(window, kAXTitleAttribute) as Any)
        print("AXMinimized", value(window, kAXMinimizedAttribute) as Any)
    } else if mode == "place" {
        var point = CGPoint(x: Double(args[4])!, y: Double(args[5])!)
        var size = CGSize(width: Double(args[6])!, height: Double(args[7])!)
        print(AXUIElementSetAttributeValue(window, kAXSizeAttribute as CFString, AXValueCreate(.cgSize, &size)!))
        print(AXUIElementSetAttributeValue(window, kAXPositionAttribute as CFString, AXValueCreate(.cgPoint, &point)!))
    } else if mode == "inspect" || mode == "press" {
        var pressed = false
        func walk(_ node: AXUIElement, _ depth: Int) {
            guard depth < 40 else { return }
            let role = value(node, kAXRoleAttribute) as? String ?? ""
            if mode == "press" {
                let label = (value(node, "AXAttributedDescription") as? NSAttributedString)?.string
                    ?? (value(node, kAXTitleAttribute) as? String)
                    ?? (value(node, kAXDescriptionAttribute) as? String)
                if !pressed, label == args[4], role == kAXButtonRole {
                    pressed = true
                    print("AXPress", label!, AXUIElementPerformAction(node, kAXPressAction as CFString).rawValue)
                }
            } else if role == kAXButtonRole || role == kAXImageRole {
                var keys: CFArray?
                AXUIElementCopyAttributeNames(node, &keys)
                print("ELEMENT", role)
                for key in keys as? [String] ?? [] where [kAXTitleAttribute,kAXDescriptionAttribute,kAXHelpAttribute,kAXValueAttribute,kAXPositionAttribute,kAXSizeAttribute,kAXEnabledAttribute,"AXAttributedDescription"].contains(key) {
                    if let attributed = value(node, key) as? NSAttributedString {
                        print(key, attributed.string)
                    } else {
                        print(key, value(node, key) as Any)
                    }
                }
            }
            for child in value(node, kAXChildrenAttribute) as? [AXUIElement] ?? [] { walk(child, depth+1) }
        }
        walk(window, 0)
    }
}
