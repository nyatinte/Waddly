#!/bin/sh
set -eu
ROOT=$(pwd)
WORK="$ROOT/dist/qa111-menu/paired"
mkdir -p "$WORK"
for SIDE in before after; do
  if [ "$SIDE" = before ]; then REV=16bf69b; else REV=e3288a5; fi
  DIR="$WORK/$SIDE"
  mkdir -p "$DIR/source"
  git archive "$REV" Sources macos | tar -x -C "$DIR/source"
  APP="$DIR/MenuProbe.app"
  mkdir -p "$APP/Contents/MacOS" "$APP/Contents/Resources"
  cp "$DIR/source/macos/Info.plist" "$APP/Contents/Info.plist"
  /usr/libexec/PlistBuddy -c 'Set :CFBundleIdentifier dev.nyatinte.waddly.qa111menu' "$APP/Contents/Info.plist"
  /usr/libexec/PlistBuddy -c 'Set :CFBundleExecutable MenuProbe' "$APP/Contents/Info.plist"
  cp -R "$DIR/source/macos/en.lproj" "$DIR/source/macos/ja.lproj" "$APP/Contents/Resources/"
  rm "$DIR/source/Sources/WaddlyApp/AppMain.swift"
  swiftc -O -swift-version 6 -I "$ROOT/dist/qa111-menu/core" -L "$ROOT/dist/qa111-menu/core" -lWaddlyCore \
    "$DIR/source/Sources/WaddlyApp/"*.swift /tmp/waddly-111-menu-render.swift \
    -o "$APP/Contents/MacOS/MenuProbe" > "$DIR/compile.log" 2>&1
  codesign --force --sign - --identifier dev.nyatinte.waddly.qa111menu "$APP"
  "$APP/Contents/MacOS/MenuProbe" "$DIR/images" > "$DIR/run.log" 2>&1
  grep 'PROBE_COMPLETED' "$DIR/run.log"
done
