import AppKit

/// Standalone native-menu renderer, not a physical-input or OS-screenshot test.
@main @MainActor struct MenuProbe {
    static var currentMenu: NSMenu?
    static var captureError: Error?
    static var captureCount = 0

    static func main() {
        let application = NSApplication.shared
        application.setActivationPolicy(.regular)
        DispatchQueue.main.async {
            do { try run() }
            catch { print("PROBE_FAILED", error); exit(1) }
            application.terminate(nil)
        }
        application.run()
    }

    static func run() throws {
        let output = URL(fileURLWithPath: CommandLine.arguments[1])
        try FileManager.default.createDirectory(at: output, withIntermediateDirectories: true)
        let domain = "Waddly111MenuProbe.\(UUID().uuidString)"
        guard let defaults = UserDefaults(suiteName: domain) else { throw failure(1) }
        defer { defaults.removePersistentDomain(forName: domain) }
        let settings = AppSettings(defaults: defaults)
        settings.registerDefaults()
        settings.showInDock = false
        settings.showInMenuBar = true
        let app = AppDelegate(settings: settings)
        app.buildPanel()
        let window = NSWindow(contentRect: NSRect(x: 200, y: 200, width: 820, height: 520), styleMask: [.titled, .closable], backing: .buffered, defer: false)
        window.title = "Isolated Waddly native-menu renderer"
        window.makeKeyAndOrderFront(nil)
        NSApp.activate(ignoringOtherApps: true)
        guard let view = window.contentView else { throw failure(2) }
        for language in [AppLanguage.english, .japanese] {
            settings.appLanguage = language
            app.buildMenu()
            for dark in [false, true] {
                let appearance = NSAppearance(named: dark ? .darkAqua : .aqua)
                window.appearance = appearance
                let root = app.statusMenu
                guard let presence = root.items.first(where: { $0.submenu?.items.contains(where: { $0.action == NSSelectorFromString("toggleDockVisibility") }) == true })?.submenu else { throw failure(3) }
                for (kind, menu) in [("status-menu", root), ("presence", presence)] {
                    let name = "\(language == .english ? "en" : "ja")-\(dark ? "dark" : "light")-\(kind)"
                    menu.appearance = appearance
                    menu.update()
                    for item in menu.items { print("ITEM", name, item.title, item.isEnabled, item.toolTip ?? "nil") }
                    try capture(menu, in: view, output: output.appendingPathComponent(name + ".png"))
                }
            }
        }
        window.orderOut(nil)
        app.panel.orderOut(nil)
        guard captureCount == 8 else { throw failure(4) }
        print("PROBE_COMPLETED", captureCount)
    }

    static func capture(_ menu: NSMenu, in view: NSView, output: URL) throws {
        currentMenu = menu
        let timer = Timer(timeInterval: 1, repeats: false) { _ in
            MainActor.assumeIsolated {
                do {
                    guard let window = NSApp.windows.first(where: { $0.isVisible && $0.level == .popUpMenu }),
                          let content = window.contentView,
                          let bitmap = content.bitmapImageRepForCachingDisplay(in: content.bounds) else { throw failure(5) }
                    content.cacheDisplay(in: content.bounds, to: bitmap)
                    guard let png = bitmap.representation(using: .png, properties: [:]) else { throw failure(6) }
                    try png.write(to: output)
                    captureCount += 1
                    print("CAPTURE", output.lastPathComponent, content.bounds.size, bitmap.pixelsWide, bitmap.pixelsHigh)
                } catch { captureError = error }
                currentMenu?.cancelTrackingWithoutAnimation()
            }
        }
        RunLoop.main.add(timer, forMode: .common)
        menu.popUp(positioning: nil, at: NSPoint(x: 40, y: 430), in: view)
        if let captureError { throw captureError }
        currentMenu = nil
    }

    static func failure(_ code: Int) -> NSError { NSError(domain: "MenuProbe", code: code) }
}
